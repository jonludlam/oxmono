Reference handling

type 'a or_ref =
  1. | Ref of string
    (*

    A $ref pointer like "#/components/schemas/Pet"

    *)
  2. | Value of 'a
    (*

    An inline value

    *)
val find_member : string -> Jsont.mem list -> Jsont.json option

Find a member by name in an object's member list

val or_ref_jsont : 'a Jsont.t -> 'a or_ref Jsont.t

Create an or_ref codec that handles $ref pointers. Uses JSON as intermediate to detect $ref field.

Schema

JSON Schema with OpenAPI extensions. We use a simplified approach where references are stored as schema or_ref.

  1. title : string option;

  3. type_ : string option;

  11. const : Jsont.json option;
  12. minimum : float option;
  13. maximum : float option;
  14. exclusive_minimum : float option;
  15. exclusive_maximum : float option;
  16. multiple_of : float option;
  17. min_length : int option;
  18. max_length : int option;
  19. pattern : string option;
  20. min_items : int option;
  21. max_items : int option;
  22. unique_items : bool;
  23. min_properties : int option;
  24. max_properties : int option;
  25. all_of : Jsont.json list option;
  26. one_of : Jsont.json list option;
  27. any_of : Jsont.json list option;
