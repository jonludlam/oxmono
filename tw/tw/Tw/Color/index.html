  3. b : int;

    *)

type oklch = {
  1. l : float;
    (*

    Lightness (0-100)

    *)
  2. c : float;
    (*

    Chroma (0-0.4+)

    *)
  3. h : float;
    (*

    Hue (0-360)

    *)
}
